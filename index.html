result.members : []; const terhapus = new Set( (Array.isArray(result.deletedMemberNos) ? result.deletedMemberNos : []).map(String) ); // Member yang sudah dihapus di server tidak boleh muncul kembali. const lokalAktif = lokal.filter( m => !terhapus.has(String(m.nomor)) ); const serverAktif = server.filter( m => !terhapus.has(String(m.nomor)) ); const gabungan = serverAktif.map(m => ({ ...m })); for (const lokalMember of lokalAktif) { const index = gabungan.findIndex( m => String(m.nomor) === String(lokalMember.nomor) ); if (index === -1) { gabungan.push({ ...lokalMember }); continue; } const serverMember = gabungan[index]; const riwayat = [ ...(Array.isArray(serverMember.riwayat) ? serverMember.riwayat : []), ...(Array.isArray(lokalMember.riwayat) ? lokalMember.riwayat : []) ]; const riwayatUnik = [ ...new Map( riwayat.map((item, i) => [ item.id || [ item.jenis, item.tanggal, item.status, i ].join('|'), item ]) ).values() ]; gabungan[index] = { ...serverMember, ...lokalMember, riwayat: riwayatUnik, totalKlaim: Math.max( Number(serverMember.totalKlaim) || 0, Number(lokalMember.totalKlaim) || 0 ) }; } for (const member of gabungan) { const simpanResponse = await fetch('/api/loyalty', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-moge-pin': pin }, body: JSON.stringify({ member }) }); if (!simpanResponse.ok) { const detail = await simpanResponse.json().catch(() => ({})); throw new Error( detail.error || 'Gagal menyimpan member ke server.' ); } } simpan(gabungan); status.textContent = 'Sinkronisasi berhasil. Total member: ' + gabungan.length; alert('Sinkronisasi berhasil. Total member: ' + gabungan.length); } catch (error) { status.textContent = 'Sinkronisasi gagal: ' + error.message; alert('Sinkronisasi gagal: ' + error.message); } }
